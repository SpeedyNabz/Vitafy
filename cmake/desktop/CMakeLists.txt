cmake_minimum_required(VERSION 3.24)
project(VitafyDesktop C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(REPOSITORY "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(SOURCE "${REPOSITORY}/build/source")
if(NOT EXISTS "${SOURCE}/src/app.c")
    message(FATAL_ERROR "Run python scripts/prepare.py first")
endif()
if(WIN32)
    find_package(SDL2 CONFIG REQUIRED)
    find_package(SDL2_image CONFIG REQUIRED)
    find_package(SDL2_ttf CONFIG REQUIRED)
    set(SDL_LIBRARIES SDL2::SDL2 SDL2_image::SDL2_image SDL2_ttf::SDL2_ttf)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL REQUIRED IMPORTED_TARGET sdl2 SDL2_image SDL2_ttf)
    set(SDL_LIBRARIES PkgConfig::SDL m)
endif()
file(GLOB_RECURSE FRONTEND CONFIGURE_DEPENDS "${SOURCE}/src/*.c")
list(FILTER FRONTEND EXCLUDE REGEX "/platform_vita\\.c$")
add_executable(VitafyPreview ${FRONTEND} "${SOURCE}/backend/mock/mock_backend.c")
target_include_directories(VitafyPreview PRIVATE "${SOURCE}/include" "${SOURCE}/src")
target_compile_definitions(VitafyPreview PRIVATE SDL_MAIN_HANDLED _CRT_SECURE_NO_WARNINGS VAM_APP_VERSION="0.1.0-preview.1-mock")
target_link_libraries(VitafyPreview PRIVATE ${SDL_LIBRARIES})
add_custom_command(TARGET VitafyPreview POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${SOURCE}/assets" "$<TARGET_FILE_DIR:VitafyPreview>/assets")
add_executable(backend_contract "${SOURCE}/backend/tests/contract_test.c" "${SOURCE}/backend/mock/mock_backend.c")
target_include_directories(backend_contract PRIVATE "${SOURCE}/include")
target_compile_definitions(backend_contract PRIVATE SDL_MAIN_HANDLED _CRT_SECURE_NO_WARNINGS)
target_link_libraries(backend_contract PRIVATE ${SDL_LIBRARIES})
add_executable(vitafy_notifications "${SOURCE}/tests/notifications_test.c" "${SOURCE}/src/notifications.c")
target_include_directories(vitafy_notifications PRIVATE "${SOURCE}/include" "${SOURCE}/src")
enable_testing()
add_test(NAME mock_premium COMMAND ${CMAKE_COMMAND} -E env --unset=VAM_MOCK_FREE --unset=VAM_MOCK_STAY_OUT VAM_MOCK_INSTANT=1 $<TARGET_FILE:backend_contract> "${REPOSITORY}/build/test-data/premium/")
add_test(NAME mock_login_transition COMMAND ${CMAKE_COMMAND} -E env --unset=VAM_MOCK_INSTANT --unset=VAM_MOCK_FREE --unset=VAM_MOCK_STAY_OUT $<TARGET_FILE:backend_contract> "${REPOSITORY}/build/test-data/login/")
add_test(NAME mock_free_account COMMAND ${CMAKE_COMMAND} -E env --unset=VAM_MOCK_STAY_OUT VAM_MOCK_INSTANT=1 VAM_MOCK_FREE=1 $<TARGET_FILE:backend_contract> "${REPOSITORY}/build/test-data/free/")
add_test(NAME notifications COMMAND vitafy_notifications)
set_tests_properties(mock_premium mock_login_transition mock_free_account notifications PROPERTIES TIMEOUT 60)
if(WIN32)
    file(GLOB_RECURSE DLLS "${REPOSITORY}/.cache/sdl/*/lib/x64/*.dll")
    foreach(APP VitafyPreview backend_contract)
        add_custom_command(TARGET ${APP} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${DLLS} "$<TARGET_FILE_DIR:${APP}>")
    endforeach()
endif()
